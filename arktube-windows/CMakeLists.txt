cmake_minimum_required(VERSION 3.21)
project(arktube_windows CXX)

if(NOT WIN32)
    message(FATAL_ERROR
        "arktube-windows only builds for Windows -- see ../arktube_linux for "
        "the Linux build, which this directory mirrors module-for-module.")
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE RelWithDebInfo)
endif()

# The WebView2 SDK ships as a NuGet package (Microsoft.Web.WebView2),
# not a system library like webkit2gtk-4.1's pkg-config file on Linux
# -- there is no Windows system package manager every target machine
# already has the way virtually every Linux desktop already has
# WebKitGTK. FetchContent pulls just the headers/libs this build needs
# out of that package at configure time, the CMake-native equivalent of
# `nuget restore` or Visual Studio's built-in NuGet integration, so a
# plain `cmake -B build -S .` still works with no extra IDE-specific
# step, matching the Linux build's own "just cmake, no extra fetch
# script" simplicity.
include(FetchContent)
FetchContent_Declare(
    webview2
    URL "https://www.nuget.org/api/v2/package/Microsoft.Web.WebView2/1.0.2592.51"
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE
)
FetchContent_MakeAvailable(webview2)

# Architecture-specific import lib -- the NuGet package ships prebuilt
# WebView2Loader.dll/.lib per architecture rather than source, since
# it's a thin loader shim around the WebView2 Runtime that's installed
# (or auto-installed) system-wide, not something this build compiles
# itself.
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
    set(ARKTUBE_WEBVIEW2_ARCH "x64")
else()
    set(ARKTUBE_WEBVIEW2_ARCH "x86")
endif()

add_executable(arktube_windows WIN32
    src/main.cpp
    src/platform/resource_paths.cpp
    src/platform/settings.cpp
    src/net/connectivity.cpp
    src/ui/overlay.cpp
    src/ui/input.cpp
    src/ui/webview.cpp
)

target_include_directories(arktube_windows PRIVATE
    src
    "${webview2_SOURCE_DIR}/build/native/include"
)
target_link_directories(arktube_windows PRIVATE
    "${webview2_SOURCE_DIR}/build/native/${ARKTUBE_WEBVIEW2_ARCH}"
)
target_compile_options(arktube_windows PRIVATE /W4 /utf-8)
target_compile_definitions(arktube_windows PRIVATE UNICODE _UNICODE)
target_link_libraries(arktube_windows PRIVATE
    WebView2Loader.dll
    gdiplus
    shcore
    shell32
    ws2_32
)

# Ship resources\ (icons, splash artwork, injected user script) and the
# WebView2 loader DLL right next to the binary -- matching the
# "next to the executable" lookup ArktubeFindResource() tries (see
# src/platform/resource_paths.h), and WebView2Loader.dll's own runtime
# requirement to sit beside whatever .exe calls it, so a plain
# `cmake --build .` is already runnable straight out of the build
# directory with no separate install step, the same goal the Linux
# build's own POST_BUILD copy_directory step states.
add_custom_command(TARGET arktube_windows POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory
            "${CMAKE_CURRENT_SOURCE_DIR}/resources"
            "$<TARGET_FILE_DIR:arktube_windows>/resources"
    COMMAND ${CMAKE_COMMAND} -E copy
            "${webview2_SOURCE_DIR}/build/native/${ARKTUBE_WEBVIEW2_ARCH}/WebView2Loader.dll"
            "$<TARGET_FILE_DIR:arktube_windows>/WebView2Loader.dll"
)

# `cmake --install .` layout: a flat directory (bin-style, no FHS
# share\ split -- see src/platform/resource_paths.h's comment on why
# Windows only needs the one "next to the executable" lookup) suitable
# for zipping or wrapping in an installer (packaging/, see that
# directory's README for the current state of actual installer
# packaging, which -- like the Linux build's AppImage -- is tracked as
# follow-up work, not yet wired into CI).
install(TARGETS arktube_windows RUNTIME DESTINATION .)
install(DIRECTORY resources/ DESTINATION resources)
install(FILES "${webview2_SOURCE_DIR}/build/native/${ARKTUBE_WEBVIEW2_ARCH}/WebView2Loader.dll"
        DESTINATION .)

# NOTE: resources/icons/arktube-spinner.png (the splash spinner, loaded
# by src/ui/overlay.cpp) is not generated by this build -- GDI+ has no
# built-in SVG decoder, unlike gdk-pixbuf's librsvg-backed loader on
# Linux (see arktube_linux/src/ui/spinner.c's fallback-path comment),
# so unlike the Linux build this one can't rasterize
# resources/icons/arktube-spinner.svg at load time. See
# packaging/README.md for the one-time (not per-build) step that
# produces the PNG from that same .svg.
